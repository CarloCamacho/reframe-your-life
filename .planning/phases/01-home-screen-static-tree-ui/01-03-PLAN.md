**src/store/appStore.js** — Create the Zustand store:
```js
import { create } from 'zustand'  // MUST be named import (v5)

export const useAppStore = create((set) => ({
  currentView: 'home',          // 'home' | 'tree'
  activeTreeId: null,            // string | null

  setView: (view) => set({ currentView: view }),
  setActiveTree: (id) => set({ activeTreeId: id, currentView: 'tree' }),
  goHome: () => set({ currentView: 'home', activeTreeId: null }),
}))
```

**src/index.css** — Replace the current content with Tailwind import + dark mode variant + theme tokens:
```css
@import "tailwindcss";

@custom-variant dark (&:where(.dark, .dark *));

@theme {
  /* Rung colour tokens */
  --color-rung-root: #a3a3a3;
  --color-rung-2-active: #3b82f6;
  --color-rung-2-dormant: #6b7280;
  --color-rung-3: #22c55e;

  /* Surface palette — calm, dark mode default */
  --color-surface: #111827;
  --color-surface-raised: #1f2937;
  --color-text-primary: #f9fafb;
  --color-text-muted: #6b7280;
}
```

These generate Tailwind utility classes: `bg-surface`, `bg-surface-raised`, `text-text-primary`, `text-text-muted`, `bg-rung-root`, `bg-rung-2-active`, `border-rung-2-active`, `text-rung-3`, etc.

**src/App.jsx** — Replace the placeholder with state-based view routing:
```jsx
import { useAppStore } from './store/appStore'
import HomeScreen from './components/Home/HomeScreen'
import TreeCanvas from './components/Tree/TreeCanvas'

export default function App() {
  const currentView = useAppStore((s) => s.currentView)

  return (
    <div className="min-h-screen bg-surface text-text-primary">
      {currentView === 'home' && <HomeScreen />}
      {currentView === 'tree' && <TreeCanvas />}
    </div>
  )
}
```
